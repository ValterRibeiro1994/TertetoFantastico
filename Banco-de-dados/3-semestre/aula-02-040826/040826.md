-- criar tabela aluno
create table tb_aluno (
	mat_aluno int not null primary key,
    nome_aluno varchar(60) not null,
    sexo_aluno enum("M", "F") not null,
    nota_aluno numeric(3,1) not null
);


-- inserir alunos na tabela
insert into tb_aluno (mat_aluno, nome_aluno, sexo_aluno, nota_aluno) values
(1312, "Ana Paula", "F", 4.5),
(1313, "João", "M", 6.8),
(1314, "Igor", "M", 7.3),
(1315, "Júlia", "F", 8);

-- filtrar alunas 
select nome_aluno, nota_aluno from tb_aluno where sexo_aluno = "F";

-- ordenar os alunos por nota 
select nome_aluno, nota_aluno from tb_aluno order by nota_aluno;

-- conta a quantidade de aluno registrado
select count(1) as qtd_aluno from tb_aluno;

-- Remover coluna da tabela
alter table tb_aluno drop COLUMN sexo_aluno;

-- criar coluna na tabela
alter table tb_aluno add status_aluno varchar(35);

-- Settar campos de coluna
update tb_aluno SET status_aluno="cursando";

update tb_aluno set status_aluno = "Aprovado" where nota_aluno > 7;
update tb_aluno set status_aluno = "Reprovado" where nota_aluno < 5;
update tb_aluno set status_aluno = "Recuperação" where nota_aluno < 7 and nota_aluno >= 5;
